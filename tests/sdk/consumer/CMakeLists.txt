cmake_minimum_required(VERSION 3.16)
project(ulk_sdk_external_consumer LANGUAGES C CXX)

set(ULK_CONSUMPTION_MODE "INSTALLED" CACHE STRING "SOURCE or INSTALLED")
set(ULK_LINKAGE "STATIC" CACHE STRING "STATIC or SHARED")
set(ULK_FIND_VERSION "1.9.0" CACHE STRING "Requested package version")
option(ULK_FIND_EXACT "Require the exact package version" ON)
option(ULK_REQUIRE_UNEXPORTED_TARGET "Exercise the unexported-target refusal" OFF)

if(ULK_CONSUMPTION_MODE STREQUAL "SOURCE")
  if(NOT DEFINED ULK_SOURCE_DIR)
    message(FATAL_ERROR "ULK_SOURCE_DIR is required in SOURCE mode")
  endif()
  add_subdirectory("${ULK_SOURCE_DIR}" ulk-provider EXCLUDE_FROM_ALL)
elseif(ULK_CONSUMPTION_MODE STREQUAL "INSTALLED")
  if(ULK_FIND_EXACT)
    find_package(UniversalLauncher ${ULK_FIND_VERSION} EXACT CONFIG REQUIRED)
  else()
    find_package(UniversalLauncher ${ULK_FIND_VERSION} CONFIG REQUIRED)
  endif()
else()
  message(FATAL_ERROR "ULK_CONSUMPTION_MODE must be SOURCE or INSTALLED")
endif()

if(ULK_REQUIRE_UNEXPORTED_TARGET)
  if(NOT TARGET UniversalLauncher::Process)
    message(FATAL_ERROR "UniversalLauncher::Process is intentionally unavailable")
  endif()
endif()

if(ULK_LINKAGE STREQUAL "STATIC")
  set(ULK_LINK_TARGET UniversalLauncher::CoreStatic)
elseif(ULK_LINKAGE STREQUAL "SHARED")
  set(ULK_LINK_TARGET UniversalLauncher::CoreShared)
else()
  message(FATAL_ERROR "ULK_LINKAGE must be STATIC or SHARED")
endif()

add_executable(ulk_sdk_external_consumer main.c)
target_link_libraries(ulk_sdk_external_consumer PRIVATE ${ULK_LINK_TARGET})
set_property(TARGET ulk_sdk_external_consumer PROPERTY C_STANDARD 99)
set_property(TARGET ulk_sdk_external_consumer PROPERTY C_STANDARD_REQUIRED ON)

add_executable(ulk_sdk_cpp_headers cpp_headers.cpp)
target_link_libraries(ulk_sdk_cpp_headers PRIVATE UniversalLauncher::Headers)
set_property(TARGET ulk_sdk_cpp_headers PROPERTY CXX_STANDARD 98)
set_property(TARGET ulk_sdk_cpp_headers PROPERTY CXX_STANDARD_REQUIRED ON)

if(MSVC)
  target_compile_definitions(ulk_sdk_external_consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(ulk_sdk_external_consumer PRIVATE /W4 /WX)
  target_compile_options(ulk_sdk_cpp_headers PRIVATE /W4 /WX)
else()
  target_compile_options(ulk_sdk_external_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(ulk_sdk_cpp_headers PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
